{{- $ratingsConfig := site.Params.ratings | default dict -}} {{- $ratingData := dict -}} {{- $buildFetch := $ratingsConfig.build_fetch | default false -}} {{- if and $buildFetch (eq .Section "blog") $ratingsConfig.api_url -}} {{- $apiUrl := strings.TrimRight "/" $ratingsConfig.api_url -}} {{- $minAverage := cast.ToFloat ($ratingsConfig.min_average | default 4.2) -}} {{- $minVotes := cast.ToInt ($ratingsConfig.min_votes | default 2) -}} {{- $slug := .RelPermalink -}} {{- $reqUrl := printf "%s/ratings?slug=%s" $apiUrl ($slug | urlquery) -}} {{- $secret := or (os.Getenv "HUGO_BUILD_SECRET") (os.Getenv "BUILD_SECRET") site.Params.build_secret -}} {{- $opts := dict "headers" (dict "X-Build-Secret" ($secret | default "")) -}} {{- with try (resources.GetRemote $reqUrl $opts) -}} {{- with .Value -}} {{- if and .Data (eq .Data.StatusCode 200) -}} {{- $content := strings.TrimLeft " \t\r\n" .Content -}} {{- if strings.HasPrefix $content "{" -}} {{- $unmarshaled := "" -}} {{- with try (transform.Unmarshal .) -}} {{- with .Value -}} {{- $unmarshaled = . -}} {{- end -}} {{- end -}} {{- if not $unmarshaled -}} {{- with try (transform.Unmarshal $content) -}} {{- with .Value -}} {{- $unmarshaled = . -}} {{- end -}} {{- end -}} {{- end -}} {{- if reflect.IsMap $unmarshaled -}} {{- $avg := cast.ToFloat (index $unmarshaled "average" | default 0) -}} {{- $votes := cast.ToInt (index $unmarshaled "votes" | default 0) -}} {{- if and (ge $avg $minAverage) (ge $votes $minVotes) -}} {{- $ratingData = dict "average" $avg "votes" $votes -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $ratingData -}}
